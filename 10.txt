import numpy as np
import tensorflow as tf
from sklearn.datasets import load_digits
from sklearn.model_selection import train_test_split

tf.random.set_seed(1)

digits = load_digits()
X = digits.data / 16.0
y = digits.target
print("Dataset shape:", X.shape, " Classes:", len(set(y)))

X_train, X_test, y_train, y_test = train_test_split(X, y, test_size=0.2, random_state=1)

model = tf.keras.Sequential([
    tf.keras.layers.Input(shape=(64,)),
    tf.keras.layers.Dense(64, activation="relu"),
    tf.keras.layers.Dense(32, activation="relu"),
    tf.keras.layers.Dense(10, activation="softmax"),
])

model.compile(optimizer="adam", loss="sparse_categorical_crossentropy", metrics=["accuracy"])
model.summary()

model.fit(X_train, y_train, epochs=15, batch_size=32, verbose=2)

loss, acc = model.evaluate(X_test, y_test, verbose=0)
print(f"\nTest loss = {loss:.4f}   Test accuracy = {acc:.4f}")

pred = model.predict(X_test[:5], verbose=0)
print("\nPredictions on 5 test samples")
for i in range(5):
    print(f"  actual = {y_test[i]}   predicted = {np.argmax(pred[i])}   confidence = {pred[i].max():.2f}")
