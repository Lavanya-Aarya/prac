import time

def dls(graph, node, goal, depth, limit, visited):
    if depth > limit:
        return None, True
    if node == goal:
        return [node], False
    visited.add(node)
    cutoff = False

    for neighbor in graph.get(node, []):
        if neighbor not in visited:
            path, hit_cutoff = dls(graph, neighbor, goal, depth + 1, limit, visited)
            if path is not None:
                visited.remove(node)
                return [node] + path, False
            if hit_cutoff:
                cutoff = True
    visited.remove(node)
    return None, cutoff

def iddfs(edges, start, goal):
    graph = {}
    for u, v in edges:
        graph.setdefault(u, []).append(v)
        graph.setdefault(v, []).append(u)
    limit = 0
    while True:
        visited = set()
        path, hit_cutoff = dls(graph, start, goal, 0, limit, visited)
        if path is not None:
            return path, "REACHABLE", limit
        if not hit_cutoff:
            return [], "UNREACHABLE", -1

        limit += 1

if __name__ == "__main__":
    n = int(input())
    edges = [tuple(input().split()) for _ in range(n)]
    start, goal = input().split()
    t0 = time.perf_counter()
    path, status, depth = iddfs(edges, start, goal)
    t1 = time.perf_counter()

    print("=== FINAL OUTPUT ===")
    if status == "REACHABLE":
        print(" ".join(path))
        print(status)
        print(f"Goal found at depth: {depth}")
    else:
        print(status)
    print(f"[IDDFS] Time: {(t1 - t0) * 1000:.4f} ms")
